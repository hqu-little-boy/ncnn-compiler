# Runtime profile follow-up: one diagnostic server-det invocation using the
# post-fix compiler. This does NOT replace the paired benchmark or performance
# gates. The archived candidate/ directory contains earlier profile artifacts;
# this new directory preserves the instrumented fusion-ID join separately.
PROFILE_EVIDENCE=${EVIDENCE}/runtime-profile/pp_ocrv5_server_det_static/candidate-fusion-site-join
PROFILE_BUILD=/tmp/ncnn-compiler-p22-validation-candidate-20260923-b
PROFILE_OUT=/tmp/p22-profile-serverdet-earlyfusion-repro
# PROFILE_OUT must be a new/empty path. ncnn-compile --profile also emits plan.
LD_LIBRARY_PATH=${PROFILE_BUILD}/lib ${PROFILE_BUILD}/tools/ncnn-compile \
  --driver ${PROFILE_BUILD}/tools/ncnn-mlir-driver \
  --opt ${PROFILE_BUILD}/bin/ncnn-mlir-opt \
  --translate /usr/lib/llvm-21/bin/mlir-translate \
  --clang /usr/lib/llvm-21/bin/clang \
  --nm /usr/lib/llvm-21/bin/llvm-nm \
  --readelf /usr/bin/llvm-readelf-21 \
  --llvm-as /usr/lib/llvm-21/bin/llvm-as \
  --param /mnt/ncnn-compiler/ncnn_modelzoo/liteocr/PP-OCRv5_server_det.param \
  --bin /mnt/ncnn-compiler/ncnn_modelzoo/liteocr/PP-OCRv5_server_det.bin \
  --model-name pp_ocrv5_server_det_static \
  --input-shape 3x640x640 --threads 6 --matmul-packing auto \
  --vector-mode fixed-width --target-feature +avx2 --target-feature +fma \
  --clang-arg=-march=x86-64-v3 --profile --emit-manifest \
  --emit-execution-plan --output-dir ${PROFILE_OUT}
python3 ${PROFILE_EVIDENCE}/run_profile_once.py \
  --library=${PROFILE_OUT}/libpp_ocrv5_server_det_static.so \
  --plan=${PROFILE_OUT}/pp_ocrv5_server_det_static.plan.json \
  --profile=/tmp/p22-serverdet-profile.ndjson \
  --perf=/tmp/p22-serverdet-diagnostic-perf.ndjson
python3 /mnt/ncnn-compiler/compiler/tools/perf_attribution_report.py \
  --perf=/tmp/p22-serverdet-diagnostic-perf.ndjson \
  --plan=${PROFILE_OUT}/pp_ocrv5_server_det_static.plan.json \
  --profile=/tmp/p22-serverdet-profile.ndjson --mode=end_to_end \
  --output=/tmp/p22-serverdet-attribution.json
